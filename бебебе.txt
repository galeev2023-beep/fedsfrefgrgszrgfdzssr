#include "sharedMemory.h"
#include <Windows.h>
#include <iostream>
#include <vector>

LPVOID pBuf = nullptr;
HANDLE hMapFile = nullptr;   // здесь это хендл канала (имя оставил прежним, чтобы не ломать closeMappingHandle)

bool connectSharedMemory() {
    static bool pipeCreated = false;

    // Храним SD и SA в static, потому что CreateNamedPipe копирует их только
    // на момент вызова, но для повторных переподключений они должны жить.
    static SECURITY_DESCRIPTOR sd;
    static SECURITY_ATTRIBUTES  sa;

    if (!pipeCreated) {
        // --- 1. Готовим разрешающую DACL ---
        // InitializeSecurityDescriptor + SetSecurityDescriptorDacl с NULL-DACL
        // даёт "полный доступ всем". Этого достаточно, чтобы клиент из-под
        // любой учётки (в т.ч. elevated) смог подключиться к пайпу.
        if (!InitializeSecurityDescriptor(&sd, SECURITY_DESCRIPTOR_REVISION)) {
            std::cerr << "[-] InitializeSecurityDescriptor failed: " << GetLastError() << "\n";
            return false;
        }

        if (!SetSecurityDescriptorDacl(&sd, TRUE, NULL, FALSE)) {
            std::cerr << "[-] SetSecurityDescriptorDacl failed: " << GetLastError() << "\n";
            return false;
        }

        sa.nLength              = sizeof(SECURITY_ATTRIBUTES);
        sa.lpSecurityDescriptor = &sd;
        sa.bInheritHandle       = FALSE;

        // --- 2. Создаём канал с этим SA ---
        //  PIPE_ACCESS_DUPLEX  — оставил как было в твоём варианте, чтобы
        //                        не ломать логику, если ты потом захочешь писать обратно.
        //  instances = 1       — один клиент (DLL в игре).
        hMapFile = CreateNamedPipe(
            L"\\\\.\\pipe\\VirelexPipe",
            PIPE_ACCESS_DUPLEX,
            PIPE_TYPE_BYTE | PIPE_READMODE_BYTE | PIPE_WAIT,
            1,
            sizeof(CoordData),   // out-buffer  (сервер → клиент)
            sizeof(CoordData),   // in-buffer   (клиент → сервер)
            0,
            &sa                  // <-- вот сюда прокидываем SECURITY_ATTRIBUTES
        );

        if (hMapFile == INVALID_HANDLE_VALUE) {
            std::cerr << "[-] CreateNamedPipe failed: " << GetLastError() << "\n";
            return false;
        }

        pipeCreated = true;
    }

    // --- 3. Ждём клиента ---
    // ConnectNamedPipe блокирующий: пока DLL внутри игры не сделает CreateFile,
    // эта функция не вернётся. Если вызываешь из главного потока рендера —
    // выноси в отдельный поток, иначе получишь фриз до запуска игры.
    printf("[cheat]: Waiting for client connection...\n");

    BOOL connected = ConnectNamedPipe(hMapFile, NULL);
    DWORD err = GetLastError();

    if (!connected && err != ERROR_PIPE_CONNECTED) {
        std::cerr << "[-] ConnectNamedPipe failed: " << err << "\n";
        DisconnectNamedPipe(hMapFile);
        return false;
    }

    // --- 4. Буфер под структуру выделяем один раз ---
    if (!pBuf) {
        pBuf = malloc(sizeof(CoordData));
        if (!pBuf) {
            std::cerr << "[-] malloc failed.\n";
            return false;
        }
        shared::sharedData = (CoordData*)pBuf;
        // на всякий случай обнулим, чтобы count не был мусором до первого чтения
        ZeroMemory(shared::sharedData, sizeof(CoordData));
    }

    // --- 5. Читаем один пакет ---
    DWORD bytesRead = 0;
    if (!ReadFile(hMapFile, shared::sharedData, sizeof(CoordData), &bytesRead, NULL)) {
        std::cerr << "[-] ReadFile failed: " << GetLastError() << "\n";
        DisconnectNamedPipe(hMapFile);
        return false;
    }

    if (bytesRead != sizeof(CoordData)) {
        std::cerr << "[-] ReadFile partial: " << bytesRead
                  << " of " << sizeof(CoordData) << "\n";
        // Считаем это неуспехом, но не рвём канал — возможно, клиент ещё пишет.
        return false;
    }

    return true;
}

void closeMappingHandle() {
    if (hMapFile && hMapFile != INVALID_HANDLE_VALUE) {
        DisconnectNamedPipe(hMapFile);
        CloseHandle(hMapFile);
        hMapFile = nullptr;
    }
    if (pBuf) {
        free(pBuf);
        pBuf = nullptr;
        shared::sharedData = nullptr;
    }
}

void getEntitiesArray(std::vector<std::pair<float, float>>& outArray) {
    if (!shared::sharedData) return;

    int count = shared::sharedData->count;
    if (count <= 0) return;
    if (count > MAX_ENTITIES) count = MAX_ENTITIES;  // страховка от рассинхрона версий

    outArray.clear();
    outArray.reserve(count);

    for (int i = 0; i < count; ++i) {
        outArray.push_back(shared::sharedData->entities[i]);
    }
}
