#include <Windows.h>
#include <iostream>
#include "options.h"
#include "MinHook.h"
#include "sharedMemory.h"

#if defined _M_X64
#pragma comment(lib, "libMinHook.x64.lib")
#elif defined _M_IX86
#pragma comment(lib, "libMinHook.x86.lib")
#endif

#include "hooks.h"

#include <GL/gl.h>
#pragma comment(lib, "OpenGL32.lib")

void cheatInitialize() {
    std::cout << "[+] cheatInitialize() started.\n";

    // --- 1. Инициализация MinHook ---
    std::cout << "[*] Initializing MinHook...\n";
    MH_STATUS mhStatus = MH_Initialize();
    if (mhStatus != MH_OK) {
        std::cout << "[-] Failed to initialize MinHook. Status code: " << mhStatus << "\n";
        return;
    }
    std::cout << "[+] MinHook initialized successfully.\n";

    // --- 2. Создание разделяемой памяти ---
    std::cout << "[*] Creating shared memory mapping...\n";
    if (!createMappingData()) {
        std::cout << "[-] createMappingData() failed. Shared memory not created.\n";
        return;
    }
    std::cout << "[+] Shared memory created successfully.\n";

    // --- 3. Получение размеров экрана ---
    options::monitorX = (float)GetSystemMetrics(SM_CXSCREEN);
    options::monitorY = (float)GetSystemMetrics(SM_CYSCREEN);
    std::cout << "[+] Screen resolution: " << options::monitorX << " x " << options::monitorY << "\n";

    // --- 4. Создание хука glDrawElements ---
    std::cout << "[*] Creating hook for glDrawElements...\n";
    mhStatus = MH_CreateHook(
        &glDrawElements,
        Hooks::get_glDrawElements(),
        reinterpret_cast<void**>(&fn_glDrawElements)
    );
    if (mhStatus != MH_OK) {
        std::cout << "[-] Failed to create hook for glDrawElements. Status code: " << mhStatus << "\n";
    }
    else {
        std::cout << "[+] Hook for glDrawElements created.\n";
    }

    // --- 5. Создание хука glFlush ---
    std::cout << "[*] Creating hook for glFlush...\n";
    mhStatus = MH_CreateHook(
        &glFlush,
        Hooks::get_glFlush(),
        reinterpret_cast<void**>(&fn_glFlush)
    );
    if (mhStatus != MH_OK) {
        std::cout << "[-] Failed to create hook for glFlush. Status code: " << mhStatus << "\n";
    }
    else {
        std::cout << "[+] Hook for glFlush created.\n";
    }

    // --- 6. Включение хука glDrawElements ---
    std::cout << "[*] Enabling hook glDrawElements...\n";
    mhStatus = MH_EnableHook(&glDrawElements);
    if (mhStatus != MH_OK) {
        std::cout << "[-] Failed to enable hook glDrawElements. Status code: " << mhStatus << "\n";
    }
    else {
        std::cout << "[+] Hook glDrawElements enabled.\n";
    }

    // --- 7. Включение хука glFlush ---
    std::cout << "[*] Enabling hook glFlush...\n";
    mhStatus = MH_EnableHook(&glFlush);
    if (mhStatus != MH_OK) {
        std::cout << "[-] Failed to enable hook glFlush. Status code: " << mhStatus << "\n";
    }
    else {
        std::cout << "[+] Hook glFlush enabled.\n";
    }

    std::cout << "[+] cheatInitialize() finished.\n";
}

void createConsole() {
    AllocConsole();

    FILE* stream;
    freopen_s(&stream, "CONOUT$", "w", stdout);
    freopen_s(&stream, "CONOUT$", "w", stderr);

    // Дублируем потоки в C++ — иначе std::cout может не подхватить консоль.
    std::ios::sync_with_stdio();
    std::cout.clear();
    std::cerr.clear();
    std::cout << "[+] Console allocated.\n";
}

bool __stdcall DllMain(HMODULE hModule, DWORD ulReason, LPVOID lpReserved) {
    if (ulReason == DLL_PROCESS_ATTACH) {
        DisableThreadLibraryCalls(hModule);

        createConsole(); // <--- раскомментировано: консоль теперь работает

        cheatInitialize();
    }
    else if (ulReason == DLL_PROCESS_DETACH) {
        MH_Uninitialize();
        closeMappingHandle();
    }

    return true;
}
