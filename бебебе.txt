#include <Windows.h>
#include <string>
#include <iostream>
#include "GL/glew.h"
#include "options.h"
#pragma comment(lib, "glew32s.lib")
#include "hooks.h"
#include "sharedMemory.h"
#include <GL/gl.h>
#pragma comment(lib, "OpenGL32.lib")
#include <glm/glm.hpp>
#include <glm/gtc/matrix_transform.hpp>
#include <glm/gtc/type_ptr.hpp>
#include <algorithm>

std::vector<std::pair<float, float>> coordsArray;

void WINAPI Hooks::hk_glDrawElements(GLenum mode, GLsizei count, GLenum type, const void* indices) {

    static bool glewInitialized = false;

    if (!glewInitialized) {
        if (glewInit() != GLEW_OK) {
            MessageBoxA(NULL, "Failed to activate GLEW", "Error", MB_ICONERROR);
        }
        glewInitialized = true;
    }

    // --- 1. Фильтр полигонов ---
    // Модели игроков и крупных энтити — count в районе 1000..150000.
    // Трава, листья, декали — меньше. mode == 4 — это GL_TRIANGLES.
    bool isReady = (count >= 1000 && count <= 150000) && (mode == 4);

    GLint currentProgram = 0;
    glGetIntegerv(GL_CURRENT_PROGRAM, &currentProgram);

    if (isReady) {
        glm::mat4 modelMatrix{ 0 };
        glm::mat4 viewMatrix{ 0 };
        glm::mat4 projectionMatrix{ 0 };

        bool gotModelMatrix = false;
        bool gotViewProjection = false;

        // --- 2. modelMatrix через юниформ ---
        if (currentProgram != 0) {
            GLint modelMatrixLocation = glGetUniformLocation(currentProgram, "modelMatrix");
            if (modelMatrixLocation != -1) {
                glGetUniformfv(currentProgram, modelMatrixLocation, glm::value_ptr(modelMatrix));
                gotModelMatrix = true;
            }
        }

        // --- 3. viewMatrix / projectionMatrix через UBO ---
        // Основной путь для шейдерного пайплайна Сталзона.
        if (currentProgram != 0) {
            GLuint blockIndexPersist = glGetUniformBlockIndex(currentProgram, "persistentBlock");
            if (blockIndexPersist != GL_INVALID_INDEX) {
                GLint bindingPoint = 0;
                glGetActiveUniformBlockiv(currentProgram, blockIndexPersist,
                                          GL_UNIFORM_BLOCK_BINDING, &bindingPoint);

                GLint gameUbo = 0;
                glGetIntegeri_v(GL_UNIFORM_BUFFER_BINDING, bindingPoint, &gameUbo);

                if (gameUbo != 0) {
                    // Сохраняем текущую привязку, чтобы вернуть её после чтения.
                    GLint previousUBO = 0;
                    glGetIntegerv(GL_UNIFORM_BUFFER_BINDING, &previousUBO);

                    glBindBuffer(GL_UNIFORM_BUFFER, gameUbo);
                    glGetBufferSubData(GL_UNIFORM_BUFFER, 0, sizeof(glm::mat4),
                                       glm::value_ptr(viewMatrix));
                    glGetBufferSubData(GL_UNIFORM_BUFFER, sizeof(glm::mat4), sizeof(glm::mat4),
                                       glm::value_ptr(projectionMatrix));
                    glBindBuffer(GL_UNIFORM_BUFFER, previousUBO);

                    // Если UBO реально вернул ненулевые матрицы — считаем успех.
                    if (viewMatrix != glm::mat4(0.0f) && projectionMatrix != glm::mat4(0.0f)) {
                        gotViewProjection = true;
                    }
                }
            }
        }

        // --- 4. Fallback через glGetFloatv ---
        // Если UBO не сработал — пробуем fixed-function стек.
        // На чисто шейдерном пайплайне вернёт identity, и мы отсеем по identity-проверке.
        if (!gotViewProjection) {
            GLfloat mvRaw[16] = { 0 };
            GLfloat pjRaw[16] = { 0 };
            glGetFloatv(GL_MODELVIEW_MATRIX, mvRaw);
            glGetFloatv(GL_PROJECTION_MATRIX, pjRaw);

            glm::mat4 mv = glm::make_mat4(mvRaw);
            glm::mat4 pj = glm::make_mat4(pjRaw);

            if (mv != glm::mat4(0.0f) && pj != glm::mat4(0.0f) &&
                mv != glm::mat4(1.0f) && pj != glm::mat4(1.0f)) {
                viewMatrix = mv;
                projectionMatrix = pj;
                gotViewProjection = true;
                // modelMatrix отсутствует в этом пути — используем identity,
                // потому что GL_MODELVIEW_MATRIX уже содержит view * model.
                if (!gotModelMatrix) {
                    modelMatrix = glm::mat4(1.0f);
                    gotModelMatrix = true;
                }
            }
        }

        if (!gotModelMatrix || !gotViewProjection) {
            goto THE_END;
        }

        // --- 5. Санити-проверки ---
        // Убрали projectionMatrix[3][3] != 0.0f — она ломала всё на Reversed-Z
        // и Ortho-проекциях (в современном Сталзоне там 1.0).
        if (modelMatrix == glm::mat4(0.0f) ||
            viewMatrix == glm::mat4(0.0f) ||
            projectionMatrix == glm::mat4(0.0f)) {
            goto THE_END;
        }

        // --- 6. Расчёт экранных координат ---
        glm::vec4 worldPos = glm::vec4(modelMatrix[3][0],
                                       modelMatrix[3][1],
                                       modelMatrix[3][2],
                                       modelMatrix[3][3]);

        // Если modelMatrix == identity (fallback-путь), worldPos = (0,0,0,1).
        // Это даст точку в центре камеры, что для шейдерного пайплайна неверно,
        // но fallback и не должен срабатывать в норме. Оставлено как страховка.
        float headHeight = 1.65f; // points at the legs, so I decided to lift it for aimbot
        worldPos.y += headHeight;

        glm::vec4 viewPosition = viewMatrix * worldPos;
        glm::vec4 clipPosition = projectionMatrix * viewPosition;

        if (clipPosition.w == 0.0f) goto THE_END;

        glm::vec4 ndcPosition = clipPosition / clipPosition.w;

        float screenX = (ndcPosition.x + 1.0f) / 2.0f * options::monitorX;
        float screenY = (1.0f - ndcPosition.y) / 2.0f * options::monitorY;

        if (screenX < options::monitorX && screenX > 0.0f &&
            screenY > 0.0f && screenY < options::monitorY) {
            coordsArray.push_back({ screenX, screenY });
        }
    }

THE_END:

    fn_glDrawElements(mode, count, type, indices);
}

void WINAPI Hooks::hk_glFlush() {

    if (!coordsArray.empty()) {
        writeEntitiesToShared(coordsArray, coordsArray.size());
        coordsArray.clear();
    }

    fn_glFlush();
}
