#include "sharedMemory.h"
#include <Windows.h>
#include <iostream>
#include <vector>

// Теперь это не хендл маппинга, а хендл канала. Имя переменной оставил
// прежним, чтобы не ломать те места, где она могла упоминаться.
HANDLE hPipe = nullptr;

// Локальный пакет под отправку. Статический — чтобы не аллоцировать 40 КБ
// на каждый кадр. Живёт в .bss, инициализируется нулями один раз.
static CoordData packet;

bool createMappingData() {
    // Оверлей-сервер мог ещё не подняться — ждём его появления.
    // До ~10 секунд (100 попыток × 100 мс) с паузой между ними.
    const int maxAttempts = 100;

    for (int attempt = 0; attempt < maxAttempts; ++attempt) {
        hPipe = CreateFile(
            L"\\\\.\\pipe\\VirelexPipe",   // тот же путь, что и в CreateNamedPipe
            GENERIC_WRITE,                 // клиент только пишет
            0,                             // без шаринга
            NULL,
            OPEN_EXISTING,                 // открываем существующий канал
            0,
            NULL
        );

        if (hPipe != INVALID_HANDLE_VALUE) {
            break;
        }

        DWORD err = GetLastError();

        // ERROR_PIPE_BUSY — сервер занят другим клиентом, ждём.
        // ERROR_FILE_NOT_FOUND — сервер ещё не создал канал, тоже ждём.
        if (err != ERROR_PIPE_BUSY && err != ERROR_FILE_NOT_FOUND) {
            std::cerr << "[-] CreateFile pipe failed, code: " << err << "\n";
            return false;
        }

        // Если канал занят — можно подождать его освобождения через WaitNamedPipe,
        // но простой Sleep тоже работает и не блокирует на неопределённый срок.
        Sleep(100);
    }

    if (hPipe == INVALID_HANDLE_VALUE || hPipe == nullptr) {
        std::cerr << "[-] Named pipe not available after retries.\n";
        hPipe = nullptr;
        return false;
    }

    // Переводим канал в блокирующий режим: WriteFile будет ждать, пока сервер
    // освободит буфер, а не возвращать ERROR_NO_DATA на каждом кадре.
    DWORD mode = PIPE_WAIT;
    if (!SetNamedPipeHandleState(hPipe, &mode, NULL, NULL)) {
        std::cerr << "[-] SetNamedPipeHandleState failed: " << GetLastError() << "\n";
        // Не критично: канал всё равно работает, просто возможны срывы при пиковой нагрузке.
    }

    std::cout << "[+] Connected to overlay pipe.\n";
    return true;
}

void closeMappingHandle() {
    if (hPipe && hPipe != INVALID_HANDLE_VALUE) {
        // Flush не нужен: WriteFile на канале синхронный, данные уже ушли.
        CloseHandle(hPipe);
        hPipe = nullptr;
    }
}

void writeEntitiesToShared(const std::vector<std::pair<float, float>>& entities, const int count) {
    if (!hPipe || hPipe == INVALID_HANDLE_VALUE) {
        return;
    }

    // Защита от переполнения: режем до MAX_ENTITIES, иначе вылезем за границы
    // массива packet.entities и получим порчу стека/краш.
    int safeCount = (count > MAX_ENTITIES) ? MAX_ENTITIES : count;

    packet.count = safeCount;

    for (int i = 0; i < safeCount; ++i) {
        packet.entities[i] = entities[i];
    }

    DWORD bytesWritten = 0;
    BOOL ok = WriteFile(
        hPipe,
        &packet,
        sizeof(CoordData),
        &bytesWritten,
        NULL
    );

    if (!ok) {
        DWORD err = GetLastError();
        std::cerr << "[-] WriteFile to pipe failed, code: " << err << "\n";

        // Канал порвался (оверлей закрылся, перезапустился, упал).
        // Закрываем хендл — следующая инициализация попробует подключиться заново.
        CloseHandle(hPipe);
        hPipe = nullptr;
    }
    else if (bytesWritten != sizeof(CoordData)) {
        // Частичная запись. Для PIPE_TYPE_BYTE это ненормально и означает,
        // что канал перешёл в message-режим или был закрыт на середине.
        std::cerr << "[-] Partial write: " << bytesWritten
                  << " of " << sizeof(CoordData) << "\n";
    }
}
